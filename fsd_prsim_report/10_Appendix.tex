\chapter{APPENDIX}

\section{Minimum Software Requirements}

\medskip
\textbf{Operating System}: Windows 10/11, macOS, or Linux
\\\\
\textbf{Runtime Environment}: Node.js (v14 or higher)
\\\\
\textbf{Package Manager}: npm or yarn
\\\\
\textbf{Database}: MongoDB (Local installation or MongoDB Atlas)
\\\\
\textbf{Web Browser}: Google Chrome / Microsoft Edge / Mozilla Firefox
\\\\
\textbf{Code Editor (for development)}: Visual Studio Code or any preferred IDE
\\\\

\section{Minimum Hardware Requirements}

\textbf{Processor}: Intel Core i3 (or AMD equivalent) or higher
\\\\
\textbf{RAM}: Minimum 4 GB (8 GB recommended for development)
\\\\
\textbf{Storage}: At least 1 GB free space for project files and dependencies
\\\\
\textbf{Display}: 1366x768 resolution or higher
\\\\
\textbf{Input Devices}: Keyboard and mouse
\\\\
\textbf{Network Connectivity}: Internet connection required for package installation and optional cloud database access
\\\\
